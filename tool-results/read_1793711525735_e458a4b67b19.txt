    23→}: Props) {

    40→    const onKey = (e: KeyboardEvent) => {
    41→      if (e.key === "Escape") onClose();

    43→    window.addEventListener("keydown", onKey);
    44→    return () => window.removeEventListener("keydown", onKey);
    45→  }, [onClose]);
    46→
    47→  return (
    48→    <AnimatePresence>
    49→      {caseStudy && (
    50→        <motion.div
    51→          initial={{ opacity: 0 }}
    52→          animate={{ opacity: 1 }}

    63→            onOpenCaseStudy={(s) => { onClose(); onSwitchCaseStudy(s); }}
    64→          />
    65→
    66→          <motion.div
    67→            id="casestudy-page-scroll"
    68→            initial={{ y: 30, opacity: 0 }}
    69→            animate={{ y: 0, opacity: 1 }}
