    22→export default function ServiceDetailModal({
    23→  slug,
    24→  onClose,
    25→  onSwitchService,
    26→  onOpenPortfolio,
    27→  onOpenCaseStudy,
    28→}: Props) {
    29→  const service = serviceDetails.find((s) => s.slug === slug);
    30→
    31→  // Lock body scroll when page is open
    32→  useEffect(() => {
    33→    if (service) {
    34→      document.body.style.overflow = "hidden";
    35→      const el = document.getElementById("service-page-scroll");
    36→      if (el) el.scrollTop = 0;
    37→    } else {
    38→      document.body.style.overflow = "";
    39→    }
    40→    return () => {
    41→      document.body.style.overflow = "";
    42→    };
    43→  }, [service]);
    44→
    45→  // ESC to close
    46→  useEffect(() => {
    47→    const onKey = (e: KeyboardEvent) => {
    48→      if (e.key === "Escape") onClose();
    49→    };
    50→    window.addEventListener("keydown", onKey);
    51→    return () => window.removeEventListener("keydown", onKey);
    52→  }, [onClose]);
    53→
    54→  return (
    55→    <AnimatePresence>
    56→      {service && (
    57→        <motion.div
    58→          initial={{ opacity: 0 }}
    59→          animate={{ opacity: 1 }}
    60→          exit={{ opacity: 0 }}
    61→          transition={{ duration: 0.3 }}
    62→          className="fixed inset-0 z-[100] bg-white"
    63→        >

    66→            variant="service"
    67→            onBack={onClose}
    68→            scrollContainerId="service-page-scroll"
    69→            onOpenService={(s) => { onClose(); onSwitchService(s); }}
    70→            onOpenPortfolio={(s) => { onClose(); onOpenPortfolio?.(s); }}
    71→            onOpenCaseStudy={(s) => { onClose(); onOpenCaseStudy?.(s); }}
    72→          />
    73→
    74→          {/* === SCROLLABLE PAGE CONTENT === */}
    75→          <motion.div
    76→            id="service-page-scroll"
    77→            initial={{ y: 30, opacity: 0 }}
    78→            animate={{ y: 0, opacity: 1 }}
    79→            exit={{ y: 30, opacity: 0 }}

    87→              onClose={onClose}
    88→            />
